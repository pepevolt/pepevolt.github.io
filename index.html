📜 Smart Contracts & Deployment Guide

1. PVLT Token Contract (ERC20)

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.19;
import "@openzeppelin/contracts/token/ERC20/ERC20.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

contract PVLTToken is ERC20, Ownable {
    constructor() ERC20("PepeVolt", "PVLT") {
        _mint(msg.sender, 1000000 * 10 ** decimals());
    }
    function mint(address to, uint256 amount) external onlyOwner {
        _mint(to, amount);
    }
}

2. PepeVolt Game Contract

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.19;
import "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

contract PepeVoltGame is Ownable {
    IERC20 public pvltToken;
    mapping(address => uint256) public energy;
    mapping(address => uint256) public maxEnergy;
    mapping(address => uint256) public gPVLT; // 100 gPVLT = 1 PVLT
    mapping(address => uint256) public lastClaim;

    event EnergyPurchased(address indexed user, uint256 amount);
    event RewardClaimed(address indexed user, uint256 amount);

    constructor(address _pvltToken) {
        pvltToken = IERC20(_pvltToken);
    }

    function buyEnergy(uint256 costPVLT) external {
        require(costPVLT == 100 * 1e18, "Cost is 100 PVLT");
        require(pvltToken.transferFrom(msg.sender, address(this), costPVLT), "Transfer failed");
        energy[msg.sender] += 10000;
        maxEnergy[msg.sender] += 10000;
        emit EnergyPurchased(msg.sender, 10000);
    }

    function claimReward(uint256 amountPVLT) external {
        require(amountPVLT >= 10 * 1e18 && amountPVLT <= 500 * 1e18, "Amount 10-500 PVLT");
        require(block.timestamp - lastClaim[msg.sender] >= 1 days, "Claim once per day");
        uint256 requiredGPVLT = (amountPVLT / 1e18) * 100;
        require(gPVLT[msg.sender] >= requiredGPVLT, "Not enough gPVLT");
        gPVLT[msg.sender] -= requiredGPVLT;
        require(pvltToken.transfer(msg.sender, amountPVLT), "Transfer failed");
        lastClaim[msg.sender] = block.timestamp;
        emit RewardClaimed(msg.sender, amountPVLT);
    }

    function addPoints(address user, uint256 points) external onlyOwner {
        gPVLT[user] += points;
    }

    function getEnergy(address user) external view returns (uint256) { return energy[user]; }
    function getMaxEnergy(address user) external view returns (uint256) { return maxEnergy[user]; }
    function getGPVLT(address user) external view returns (uint256) { return gPVLT[user]; }
}

🚀 Deployment Steps (Polygon)

  1. Install Node.js, Hardhat or Remix IDE
  2. Deploy PVLTToken.sol first → copy contract address
  3. Deploy PepeVoltGame.sol with PVLT token address as constructor param
  4. Transfer some PVLT tokens to Game contract (for claiming rewards)
  5. Set GAME_CONTRACT_ADDRESS and PVLT_TOKEN_ADDRESS in the HTML above

📦 Hardhat Deployment Script

// deploy.js
const hre = require("hardhat");

async function main() {
    const PVLT = await hre.ethers.getContractFactory("PVLTToken");
    const pvlt = await PVLT.deploy();
    await pvlt.deployed();
    console.log("PVLT deployed to:", pvlt.address);

    const Game = await hre.ethers.getContractFactory("PepeVoltGame");
    const game = await Game.deploy(pvlt.address);
    await game.deployed();
    console.log("Game deployed to:", game.address);
}
main().catch(console.error);

⚠️ Important Notes:

  • After deployment, call game.addPoints(userAddress, gpvltAmount) to sync user gPVLT
  • Or integrate backend API to submit point accumulations
  • Daily claim limit: 10-500 PVLT per wallet
  • Buy energy costs 100 PVLT → +10,000 energy & +10,000 max energy
